Sheet 02 — Sign-In & Sign-Up Verificationtest_auth.py
import pytest
from appium.webdriver.common.appiumby import AppiumBy
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

def test_signin_screen_components_load(driver):
    """Verify the initial SignInView and its critical interactive components."""
    wait = WebDriverWait(driver, 15)

    # Scope the header to StaticText to avoid matching the button.
    header_predicate = 'type == "XCUIElementTypeStaticText" AND label == "Sign in"'
    header = wait.until(
        EC.visibility_of_element_located((AppiumBy.IOS_PREDICATE, header_predicate))
    )
    assert header.is_displayed(), "Sign in screen header is not visible."

    button_predicate = 'type == "XCUIElementTypeButton" AND label == "Sign in"'
    signin_button = wait.until(
        EC.element_to_be_clickable((AppiumBy.IOS_PREDICATE, button_predicate))
    )
    assert signin_button.is_enabled(), "Primary 'Sign in' button is disabled."

    footer_predicate = 'label CONTAINS "Don\'t have an account?"'
    footer_prompt = wait.until(
        EC.visibility_of_element_located((AppiumBy.IOS_PREDICATE, footer_predicate))
    )
    assert footer_prompt.is_displayed(), "Footer sign-up link prompt is not visible."
    print("\n✅ Day 1 Session Successful: StageTimePNW SignInView fully verified!")


def test_successful_signup_flow(driver):
    """Verify a new user can sign up and reach the welcome screen."""
    wait = WebDriverWait(driver, 15)

    email_predicate = 'type == "XCUIElementTypeTextField" AND name == "email_input"'
    email_field = wait.until(
        EC.visibility_of_element_located((AppiumBy.IOS_PREDICATE, email_predicate))
    )
    email_field.send_keys("[email protected]")

    password_predicate = 'type == "XCUIElementTypeSecureTextField" AND name == "password_input"'
    password_field = wait.until(
        EC.visibility_of_element_located((AppiumBy.IOS_PREDICATE, password_predicate))
    )
    password_field.send_keys("TestPass123!")

    signup_predicate = 'type == "XCUIElementTypeButton" AND label == "Sign up"'
    signup_button = wait.until(
        EC.element_to_be_clickable((AppiumBy.IOS_PREDICATE, signup_predicate))
    )
    signup_button.click()

    welcome_predicate = 'type == "XCUIElementTypeStaticText" AND label == "Welcome"'
    welcome_header = wait.until(
        EC.visibility_of_element_located((AppiumBy.IOS_PREDICATE, welcome_predicate))
    )
    assert welcome_header.is_displayed(), "Welcome screen header is not visible after sign-up."
    print("\n✅ Day 1 Session Successful: StageTimePNW Sign-Up Flow fully verified!")
